getSubscriptionInfo(name, x__xgafv=None) 
Gets the aggregated subscription and license information for a given customer as a singleton resource. Note: Because the data is snapshotted for ChromeOS devices, the exact numbers can be delayed by up to 10 seconds.

Args:
  name: string, Required. The resource name of the customer subscription info to retrieve. Format: customers/{customer}/subscriptionInfo (required)
  x__xgafv: string, V1 error format.
    Allowed values
      1 - v1 error format
      2 - v2 error format

Returns:
  An object of the form:

    { # Singleton resource representing the subscription and license information for a given customer.
  "chromeEnterpriseUpgradeAnnual": { # Information for annually recurring subscriptions/licenses. # Output only. Details for Chrome Enterprise Upgrade Annual SKU.
    "counts": { # Resource counts for a specific license type. # Output only. License counts for this subscription.
      "assignedCount": "A String", # Output only. Total number of licenses currently assigned to users or devices.
      "totalPurchasedCount": "A String", # Output only. Total number of licenses purchased.
      "unassignedCount": "A String", # Output only. Total number of licenses available for assignment.
    },
    "subscriptionEndTime": "A String", # Output only. The date when the current subscription period ends.
  },
  "chromePerpetualUpgrade": { # Information for perpetual licenses. # Output only. Details for Chrome Ent/Edu Upgrade Perpetual SKU.
    "conditionallyAvailableCount": "A String", # Output only. Number of standalone perpetual upgrades that were previously enrolled and later unenrolled for RMA, warranty, or repair reasons, and are only available to be reused on a same model device. Conditionally available licenses are a subset of the Assigned total.
    "counts": { # Resource counts for a specific license type. # Output only. License counts for the perpetual SKU.
      "assignedCount": "A String", # Output only. Total number of licenses currently assigned to users or devices.
      "totalPurchasedCount": "A String", # Output only. Total number of licenses purchased.
      "unassignedCount": "A String", # Output only. Total number of licenses available for assignment.
    },
    "retiredCount": "A String", # Output only. Number of perpetual upgrades that were purchased but are no longer available for enrollment. These are perpetual upgrades that were enrolled to a device that has since been retired, and the upgrade cannot be transferred or re-used on a different device. Retired licenses are a subset of the Assigned total.
  },
  "kioskSignageUpgradeAnnual": { # Information for annually recurring subscriptions/licenses. # Output only. Details for Kiosk & Signage Upgrade Annual SKU.
    "counts": { # Resource counts for a specific license type. # Output only. License counts for this subscription.
      "assignedCount": "A String", # Output only. Total number of licenses currently assigned to users or devices.
      "totalPurchasedCount": "A String", # Output only. Total number of licenses purchased.
      "unassignedCount": "A String", # Output only. Total number of licenses available for assignment.
    },
    "subscriptionEndTime": "A String", # Output only. The date when the current subscription period ends.
  },
  "name": "A String", # Identifier. The resource name of the customer subscription info. Format: customers/{customer}/subscriptionInfo
}
