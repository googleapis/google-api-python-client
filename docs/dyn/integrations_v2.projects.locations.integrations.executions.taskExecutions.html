get(name, x__xgafv=None) 
Get a TaskExecution in the specified project.

Args:
  name: string, Required. The TaskExecution to retrieve. Format: projects/{project}/locations/{location}/integrations/{integration}/executions/{execution}/taskExecutions/{task_execution} (required)
  x__xgafv: string, V1 error format.
    Allowed values
      1 - v1 error format
      2 - v2 error format

Returns:
  An object of the form:

    { # Execution of a single task within an integration
  "name": "A String", # Identifier. Task execution resource name.
  "taskExecutionDetails": [ # Details of the task execution.
    { # Details of the task execution.
      "taskAttemptStats": [ # List for the current task execution attempts.
        { # Status for the execution attempt.
          "endTime": "A String", # The end time of the execution for the current attempt.
          "startTime": "A String", # The start time of the execution for the current attempt. This could be in the future if it's been scheduled.
        },
      ],
      "taskExecutionState": "A String", # Output only. The execution state of this task.
      "taskNumber": "A String", # Pointer to the task config it used for execution.
    },
  ],
  "taskExecutionMetadata": { # Metadata of the task execution. # Optional. Metadata of the task execution.
    "ancestorIterationNumbers": [ # Optional. Ancestor iteration number for the task (it will only be non-empty if the task is under 'private integration').
      "A String",
    ],
    "ancestorTaskNumbers": [ # Optional. Ancestor task number for the task (it will only be non-empty if the task is under 'private integration').
      "A String",
    ],
    "executionAttempt": 42, # The execution attempt number this execution belongs to.
    "privateIntegrationName": "A String", # Optional. The direct integration which the execution belongs to.
    "task": "A String", # The task name associated with this execution.
    "taskAttempt": 42, # The task attempt number this execution belongs to.
    "taskLabel": "A String", # The task label associated with this execution.
    "taskNumber": "A String", # The task number associated with this execution.
  },
  "variables": { # Optional. Variables used during the execution.
    "a_key": "", # Properties of the object.
  },
}
